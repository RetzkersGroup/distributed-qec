% Expansion of `MXX 0 1` via two ancilla qubits.
%
% Implements the joint XX measurement as:
%     RX 2
%     R  3
%     CX 2 3
%     CX 2 0
%     CX 3 1
%     MX 2
%     MX 3
%
% Wires are drawn in the order q_0, q_2, q_3, q_1 so the data-ancilla
% CXs (CX 2 0 and CX 3 1) are short, non-crossing vertical lines.
%
% Compile with:
%     pdflatex mxx_quantikz.tex

\documentclass[border=2pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{quantikz2}

\begin{document}
\begin{quantikz}
  \lstick{$q_0$}     & \qw       & \targ{}   & \qw       & \qw        & \rstick{$q_0$} \\
  \lstick{$\ket{+}$} & \ctrl{1}  & \ctrl{-1} & \qw       & \meter{X}  & \\
  \lstick{$\ket{0}$} & \targ{}   & \qw       & \ctrl{1}  & \meter{X}  & \\
  \lstick{$q_1$}     & \qw       & \qw       & \targ{}   & \qw        & \rstick{$q_1$}
\end{quantikz}
\end{document}
